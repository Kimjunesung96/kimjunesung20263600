def knightMovable():
    # 현재 나이트의 위치 입력받기
    input_data = input("현재 나이트의 위치를 입력하세요: ")
    row = int(input_data[1])
    
    # 가로 위치에 대한 숫자 변환
    column = int(ord(input_data[0])) - int(ord('a')) + 1

    # 나이트가 이동할 수 있는 8가지 방향 정의
    steps = [(-2, -1), (-1, -2), (1, -2), (2, -1), (2, 1), (1, 2), (-1, 2), (-2, 1)]

    # 8가지 방향에 대하여 각 위치로 이동이 가능한지 확인
    result = 0
    for step in steps:
        # 이동하고자 하는 위치 확인
        next_row = row + step[0]
        next_column = column + step[1]
        # 해당 위치로 이동이 가능하다면 카운트 증가
        if next_row >= 1 and next_row <= 8 and next_column >= 1 and next_column <= 8:
            result += 1

    print(f"이동 가능한 수 = {result}")

def moveLRUD():
    # N 입력받기
    n = int(input("공간의 가로/세로 크기를 입력하세요: "))
    x, y = 1, 1
    plans = input("이동 계획(L, R, U, D)을 순서대로 입력하세요: ").split()

    # L, R, U, D에 따른 이동 방향
    dx = [0, 0, -1, 1]
    dy = [-1, 1, 0, 0]
    move_types = ['L', 'R', 'U', 'D']

    # 이동 계획을 하나씩 확인
    for plan in plans:
        # 이동 후 좌표 구하기
        for i in range(len(move_types)):
            if plan == move_types[i]:
                nx = x + dx[i]
                ny = y + dy[i]
        # 공간을 벗어나는 경우 무시
        if nx < 1 or ny < 1 or nx > n or ny > n:
            continue
        # 이동 수행
        x, y = nx, ny

    print(f"최종 도착 위치는 ({x}, {y}) 입니다.")

    while queue:

def dfs_ice(graph, n, m, x, y):
    
    if x <= -1 or x >= n or y <= -1 or y >= m:
        return False

    if graph[x][y] == 0:
        graph[x][y] = 1

        dfs_ice(graph, n, m, x - 1, y)
        dfs_ice(graph, n, m, x, y - 1)
        dfs_ice(graph, n, m, x + 1, y)
        dfs_ice(graph, n, m, x, y + 1)

        return True

    return False
